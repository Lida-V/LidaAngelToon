// LidaAngelToon: N.L with the key-light shadow (Unlit in Unreal: no shadow map, Atten = 1)
float l2 = dot(Lw, Lw);
float3 l = (l2 > 0.0) ? Lw * rsqrt(l2) : float3(0.0, 0.0, 1.0);
return clamp(dot(Nw, l), 0.0, 1.0) * Atten * Gain;
