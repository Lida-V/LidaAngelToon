// LidaAngelToon: shading normal = vertex normal, flipped toward the viewer on back faces
float3 n = Nv;
float l2 = dot(n, n);
n = (l2 > 0.0) ? n * rsqrt(l2) : float3(0.0, 0.0, 1.0);
return n * ((Sgn < 0.0) ? -1.0 : 1.0);
